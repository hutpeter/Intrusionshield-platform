0430: function formatSlot(
0431:     slot: AvailabilitySlot,
0432:     locale: string,
0433:     timeZone: string

0435:     const start = new Date(slot.start);

0438:     if (
0439:         Number.isNaN(start.getTime()) ||
0440:         Number.isNaN(end.getTime())

0442:         return `${slot.start} - ${slot.end}`;
0443:     }
0444: 
0445:     const formatter =
0446:         new Intl.DateTimeFormat(locale, {
0447:             month: "short",
0448:             day: "numeric",
0449:             hour: "numeric",
0450:             minute: "2-digit",
0451:             timeZone
0452:         });
0453: 
0454:     const timeFormatter =
0455:         new Intl.DateTimeFormat(locale, {
0456:             hour: "numeric",
0457:             minute: "2-digit",
0458:             timeZone
0459:         });
0460: 
0461:     return `${formatter.format(start)} - ${timeFormatter.format(end)}`;
